# 2、鉴权

```python
def md5_encrypt(text):
    # 创建md5对象
    md5_obj = hashlib.md5()
    # 更新数据，注意这里需要将字符串转换为字节
    md5_obj.update(text.encode('utf-8'))
    # 获取加密后的字符串
    encrypted_text = md5_obj.hexdigest()
    return encrypted_text


def get_linux_pwd():
    now_date = Utils.get_now_date()
    text = '5ncp31:' + now_date
    return Utils.md5_encrypt(text)[8:24]
```

('
awk -F ''.'' -v start="${start_time}" -v end="${end_time}" ''
{
    log_time = $1
    log_time_seconds = mktime(gensub(/:/," ", "g", gensub(/-/," ", "g", $1)))
    start_seconds = mktime(gensub(/:/," ", "g", gensub(/-/," ", "g", start)))
    end_seconds = mktime(gensub(/:/," ", "g", gensub(/-/," ", "g", end)))
